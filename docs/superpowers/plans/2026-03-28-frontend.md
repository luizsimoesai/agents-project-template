# Frontend — Chat App com Aegra + LangGraph

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar o frontend completo em `frontend/` — Next.js 15 com autenticação JWT, CRUD de assistentes, chat com streaming SSE via `useStream`, e painel admin.

**Architecture:** Sidebar dupla (ícones + painel contextual). TanStack Query gerencia estado de servidor (CRUD). Zustand gerencia somente auth (access token + user). `useStream` do `@langchain/langgraph-sdk` chama diretamente o endpoint nativo do Aegra (`POST /threads/{id}/runs/stream`) com `Authorization: Bearer` nos headers. Middleware Next.js lê cookie `access_token` para proteger rotas.

**Tech Stack:** Next.js 15 (App Router), TypeScript, Tailwind CSS, shadcn/ui, @tanstack/react-query, @langchain/langgraph-sdk, zustand, next-themes, sonner, zod, react-hook-form, Vitest + React Testing Library.

```ts
import { create } from "zustand"
import type { UserRead } from "@/lib/types"

const COOKIE_NAME = "access_token"

function setCookie(value: string) {
  document.cookie = `${COOKIE_NAME}=${value};path=/;SameSite=Lax`
}

function removeCookie() {
  document.cookie = `${COOKIE_NAME}=;path=/;expires=Thu, 01 Jan 1970 00:00:00 GMT`
}

interface AuthStore {
  accessToken: string | null
  user: UserRead | null
  setAuth: (token: string, user: UserRead) => void
  clearAuth: () => void
}

export const useAuthStore = create<AuthStore>((set) => ({
  accessToken: null,
  user: null,
  setAuth: (token, user) => {
    setCookie(token)
    set({ accessToken: token, user })
  },
  clearAuth: () => {
    removeCookie()
    localStorage.removeItem("refresh_token")
    set({ accessToken: null, user: null })
  },
}))
```

```ts
import { useAuthStore } from "@/lib/store/auth"

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:2024"

let isRefreshing = false
let refreshPromise: Promise<boolean> | null = null

async function tryRefresh(): Promise<boolean> {
  if (isRefreshing) return refreshPromise!
  isRefreshing = true
  refreshPromise = (async () => {
    const refreshToken = localStorage.getItem("refresh_token")
    if (!refreshToken) return false
    try {
      const res = await fetch(`${API_URL}/api/v1/auth/refresh`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ refresh_token: refreshToken }),
      })
      if (!res.ok) return false
      const data = await res.json()
      const { user } = useAuthStore.getState()
      useAuthStore.getState().setAuth(data.access_token, user!)
      localStorage.setItem("refresh_token", data.refresh_token)
      return true
    } catch {
      return false
    } finally {
      isRefreshing = false
      refreshPromise = null
    }
  })()
  return refreshPromise
}

export async function apiClient(
  path: string,
  options: RequestInit = {}
): Promise<Response> {
  const { accessToken } = useAuthStore.getState()

  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...(options.headers as Record<string, string>),
  }

  if (accessToken) {
    headers["Authorization"] = `Bearer ${accessToken}`
  }

  const res = await fetch(`${API_URL}${path}`, { ...options, headers })

  if (res.status === 401) {
    const refreshed = await tryRefresh()
    if (refreshed) {
      const { accessToken: newToken } = useAuthStore.getState()
      if (newToken) headers["Authorization"] = `Bearer ${newToken}`
      return fetch(`${API_URL}${path}`, { ...options, headers })
    }
    useAuthStore.getState().clearAuth()
    if (typeof window !== "undefined") {
      window.location.href = "/login"
    }
  }

  return res
}
```

```ts
import { apiClient } from "./client"
import { useAuthStore } from "@/lib/store/auth"
import type { TokenResponse, UserRead, UserCreate } from "@/lib/types"

export async function login(email: string, password: string): Promise<void> {
  const res = await apiClient("/api/v1/auth/login", {
    method: "POST",
    body: JSON.stringify({ email, password }),
  })
  if (!res.ok) {
    const err = await res.json()
    throw new Error(
      typeof err.detail === "string" ? err.detail : "Credenciais inválidas"
    )
  }
  const data: TokenResponse = await res.json()
  localStorage.setItem("refresh_token", data.refresh_token)
  const user = await getMe(data.access_token)
  useAuthStore.getState().setAuth(data.access_token, user)
}

export async function register(data: UserCreate): Promise<UserRead> {
  const res = await apiClient("/api/v1/auth/register", {
    method: "POST",
    body: JSON.stringify(data),
  })
  if (!res.ok) {
    const err = await res.json()
    throw new Error(
      typeof err.detail === "string" ? err.detail : "Erro ao criar conta"
    )
  }
  return res.json()
}

async function getMe(token: string): Promise<UserRead> {
  const res = await fetch(
    `${process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:2024"}/api/v1/auth/me`,
    { headers: { Authorization: `Bearer ${token}` } }
  )
  if (!res.ok) throw new Error("Falha ao buscar perfil")
  return res.json()
}

export async function refreshSession(): Promise<void> {
  const refreshToken = localStorage.getItem("refresh_token")
  if (!refreshToken) throw new Error("No refresh token")
  const res = await apiClient("/api/v1/auth/refresh", {
    method: "POST",
    body: JSON.stringify({ refresh_token: refreshToken }),
  })
  if (!res.ok) throw new Error("Refresh failed")
  const data: TokenResponse = await res.json()
  localStorage.setItem("refresh_token", data.refresh_token)
  const user = await getMe(data.access_token)
  useAuthStore.getState().setAuth(data.access_token, user)
}
```

```ts
import { NextResponse } from "next/server"
import type { NextRequest } from "next/server"

const PUBLIC_PATHS = ["/login", "/signup"]

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl
  const isPublic = PUBLIC_PATHS.some((p) => pathname.startsWith(p))
  const token = request.cookies.get("access_token")?.value

  if (!isPublic && !token) {
    return NextResponse.redirect(new URL("/login", request.url))
  }

  if (isPublic && token) {
    return NextResponse.redirect(new URL("/assistants", request.url))
  }

  return NextResponse.next()
}

export const config = {
```

```tsx
import { useAuthStore } from "@/lib/store/auth"
import { refreshSession } from "@/lib/api/auth"

export function AuthProvider({ children }: { children: React.ReactNode }) {
  useEffect(() => {
    // Se há cookie mas não há token no Zustand (ex: reload de página),
    // tenta revalidar via refresh token
    const { accessToken } = useAuthStore.getState()
    if (!accessToken) {
      const refreshToken = localStorage.getItem("refresh_token")
      if (refreshToken) {
        refreshSession().catch(() => useAuthStore.getState().clearAuth())
      }
    }
  }, [])

  return <>{children}</>
}
```

```tsx
    setValue("")
    ref.current?.focus()

        ref={ref}
        value={value}
        onChange={(e) => setValue(e.target.value)}
        onKeyDown={handleKeyDown}
        placeholder="Digite uma mensagem... (Enter para enviar, Shift+Enter para nova linha)"
        disabled={disabled || isStreaming}
        className={cn("resize-none min-h-[40px] max-h-[160px] font-mono text-sm", "flex-1")}
```

```tsx
"use client"

import { useEffect, useRef } from "react"
import { useStream } from "@langchain/langgraph-sdk/react"
import { toast } from "sonner"
import { ChatMessage } from "./ChatMessage"
import { ChatInput } from "./ChatInput"
import { useAuthStore } from "@/lib/store/auth"
import { useThreadHistory } from "@/lib/hooks/useThreads"
import { Skeleton } from "@/components/ui/skeleton"
import type { AssistantRead } from "@/lib/types"

interface ChatViewProps {
  threadId: string
  assistant: AssistantRead
}

export function ChatView({ threadId, assistant }: ChatViewProps) {
  const accessToken = useAuthStore((s) => s.accessToken)
  const bottomRef = useRef<HTMLDivElement>(null)
  const apiUrl = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:2024"

  const { data: history, isLoading: historyLoading } = useThreadHistory(threadId)

  const { messages, submit, isStreaming, stop } = useStream<{
    messages: Array<{ role: string; content: string }>
  }>({
    apiUrl,
    assistantId: assistant.assistant_id,
    threadId,
    defaultHeaders: accessToken
      ? { Authorization: `Bearer ${accessToken}` }
      : {},
    messagesKey: "messages",
    onError: (err) => {
      toast.error("Erro no streaming: " + (err?.message ?? "Conexão perdida"))
    },
  })

  // Scroll to bottom on new messages
  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth" })
  }, [messages, history])

  function handleSubmit(content: string) {
    submit({ messages: [{ role: "human", content }] })
  }

  // Filter out history messages already reflected in useStream messages
  const historyToShow = history ?? []
  const streamMessages = messages ?? []

  // When streaming has messages, stop showing history to avoid duplicates
  // (useStream manages the full conversation from the LangGraph server)
  const displayMessages =
    streamMessages.length > 0
      ? streamMessages
      : historyToShow.map((m) => ({ role: m.role, content: m.content }))

  return (
    <div className="flex flex-col h-full">
      {/* Messages area */}
      <div className="flex-1 overflow-y-auto px-6 py-4">
        {historyLoading && streamMessages.length === 0 ? (
          <div className="space-y-4">
            {Array.from({ length: 3 }).map((_, i) => (
              <Skeleton key={i} className="h-12 w-full" />
            ))}
          </div>
        ) : displayMessages.length === 0 ? (
          <div className="h-full flex items-center justify-center text-muted-foreground">
            <p className="font-mono text-sm">inicie a conversa abaixo</p>
          </div>
        ) : (
          displayMessages.map((msg, i) => (
            <ChatMessage
              key={i}
              role={msg.role}
              content={typeof msg.content === "string" ? msg.content : ""}
              isStreaming={
                isStreaming &&
                i === displayMessages.length - 1 &&
                (msg.role === "ai" || msg.role === "assistant")
              }
            />
          ))
        )}
        <div ref={bottomRef} />
      </div>

      {/* Input */}
      <ChatInput
        onSubmit={handleSubmit}
        onStop={stop}
        isStreaming={isStreaming}
      />
    </div>
  )
}
```

```tsx
"use client"

import { useState } from "react"
import { toast } from "sonner"
import { Trash2, ShieldCheck } from "lucide-react"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Badge } from "@/components/ui/badge"
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
} from "@/components/ui/table"
import {
  AlertDialog, AlertDialogAction, AlertDialogCancel,
  AlertDialogContent, AlertDialogDescription, AlertDialogFooter,
  AlertDialogHeader, AlertDialogTitle, AlertDialogTrigger,
} from "@/components/ui/alert-dialog"
import { Skeleton } from "@/components/ui/skeleton"
import {
  useUsers, useActivateUser, useDeactivateUser, useDeleteUser,
} from "@/lib/hooks/useUsers"
import { useAuthStore } from "@/lib/store/auth"

type StatusFilter = "all" | "active" | "inactive"

export function AdminUserTable() {
  const [search, setSearch] = useState("")
  const [filter, setFilter] = useState<StatusFilter>("all")
  const currentUser = useAuthStore((s) => s.user)

  const { data: users, isLoading } = useUsers()
  const activateUser = useActivateUser()
  const deactivateUser = useDeactivateUser()
  const deleteUser = useDeleteUser()

  const filtered = (users ?? [])
    .filter((u) => {
      if (filter === "active") return u.is_active
      if (filter === "inactive") return !u.is_active
      return true
    })
    .filter((u) =>
      search ? u.name?.toLowerCase().includes(search.toLowerCase()) : true
    )

  async function handleToggleActive(id: string, isActive: boolean) {
    try {
      if (isActive) {
        await deactivateUser.mutateAsync(id)
        toast.success("Usuário desativado")
      } else {
        await activateUser.mutateAsync(id)
        toast.success("Usuário ativado")
      }
    } catch (err) {
      toast.error(err instanceof Error ? err.message : "Erro")
    }
  }

  async function handleDelete(id: string) {
    try {
      await deleteUser.mutateAsync(id)
      toast.success("Usuário apagado")
    } catch (err) {
      toast.error(err instanceof Error ? err.message : "Erro ao apagar usuário")
    }
  }

  return (
    <div className="space-y-4">
      {/* Filters */}
      <div className="flex gap-2 items-center flex-wrap">
        <Input
          placeholder="buscar por nome..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="max-w-[240px] font-mono text-sm"
        />
        {(["all", "active", "inactive"] as StatusFilter[]).map((f) => (
          <Button
            key={f}
            variant={filter === f ? "default" : "outline"}
            size="sm"
            className="font-mono text-xs"
            onClick={() => setFilter(f)}
            style={filter === f ? { backgroundColor: "#136ddd" } : {}}
          >
            {f === "all" ? "todos" : f === "active" ? "ativos" : "inativos"}
          </Button>
        ))}
      </div>

      {/* Table */}
      <div className="border border-border rounded-md overflow-hidden">
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead className="font-mono text-xs">nome</TableHead>
              <TableHead className="font-mono text-xs">email</TableHead>
              <TableHead className="font-mono text-xs">status</TableHead>
              <TableHead className="font-mono text-xs">criado em</TableHead>
              <TableHead className="font-mono text-xs w-[80px]">ações</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {isLoading
              ? Array.from({ length: 4 }).map((_, i) => (
                  <TableRow key={i}>
                    {Array.from({ length: 5 }).map((_, j) => (
                      <TableCell key={j}><Skeleton className="h-4 w-full" /></TableCell>
                    ))}
                  </TableRow>
                ))
              : filtered.map((u) => (
                  <TableRow key={u.id} className={u.id === currentUser?.id ? "bg-accent/30" : ""}>
                    <TableCell className="font-medium">
                      <div className="flex items-center gap-1">
                        {u.name ?? "—"}
                        {u.is_admin && <ShieldCheck size={12} style={{ color: "#136ddd" }} />}
                      </div>
                    </TableCell>
                    <TableCell className="font-mono text-xs">{u.email}</TableCell>
                    <TableCell>
                      <Badge
                        variant={u.is_active ? "outline" : "secondary"}
                        className="font-mono text-xs cursor-pointer"
                        onClick={() => handleToggleActive(u.id, u.is_active)}
                      >
                        {u.is_active ? "ativo" : "inativo"}
                      </Badge>
                    </TableCell>
                    <TableCell className="font-mono text-xs text-muted-foreground">
                      {new Date(u.created_at).toLocaleDateString("pt-BR")}
                    </TableCell>
                    <TableCell>
                      <AlertDialog>
                        <AlertDialogTrigger asChild>
                          <Button
                            variant="ghost"
                            size="icon"
                            className="h-7 w-7 text-destructive hover:bg-destructive/10"
                            disabled={u.id === currentUser?.id}
                          >
                            <Trash2 size={12} />
                          </Button>
                        </AlertDialogTrigger>
                        <AlertDialogContent>
                          <AlertDialogHeader>
                            <AlertDialogTitle className="font-mono">apagar usuário?</AlertDialogTitle>
                            <AlertDialogDescription>
                              Esta ação é permanente. O usuário <strong>{u.email}</strong> será removido.
                            </AlertDialogDescription>
                          </AlertDialogHeader>
                          <AlertDialogFooter>
                            <AlertDialogCancel>cancelar</AlertDialogCancel>
                            <AlertDialogAction
                              onClick={() => handleDelete(u.id)}
                              className="bg-destructive hover:bg-destructive/90"
                            >
                              apagar
                            </AlertDialogAction>
                          </AlertDialogFooter>
                        </AlertDialogContent>
                      </AlertDialog>
                    </TableCell>
                  </TableRow>
                ))}
          </TableBody>
        </Table>
      </div>
      <p className="text-xs text-muted-foreground font-mono">
        {filtered.length} usuário{filtered.length !== 1 ? "s" : ""}
      </p>
    </div>
  )
}
```
